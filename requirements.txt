# Ohbot Requirements
# Install with: pip install -r requirements.txt

# Web servers (launcher, GUI, chatbot API)
flask>=3.0.0

# AI chat
openai>=1.0.0

# HTTP client used by ohbot_chat.py to talk to the chatbot server
httpx>=0.27.0

# Azure Speech — text-to-speech and speech recognition
azure-cognitiveservices-speech>=1.38.0

# Loads API keys from the .env file
python-dotenv>=1.0.0

# Ohbot hardware library.
#
# PINNED ON PURPOSE. This is the exact version the robot has been running on
# and is known to work — checked on the old Pi 2026-08-07. The line used to
# say ">=2.0.6", meaning "or anything newer", which would have let a build a
# year from now silently pick up an ohbot 5.x with different motor behaviour
# and no clue as to why the head stopped moving.
#
# To move to a newer version, change the number here deliberately, then test
# the motors before deploying. Don't loosen it back to ">=".
ohbot==4.0.18

# Talks to the Ohbot brain board over the USB serial cable.
# Imported directly by yobot_core.py and center_servos.py, so it must be
# listed here in its own right. The 'ohbot' package above happens to pull it
# in as well, which is why this was never noticed — but relying on someone
# else's dependency list is how builds break silently later.
pyserial>=3.5

# Raspberry Pi GPIO (for the wake button — skipped gracefully if not available)
#
# PI ONLY. The tail of this line is a marker meaning "only install this on a
# Raspberry Pi". Without it, `pip install -r requirements.txt` FAILS on
# Windows and Mac — RPi.GPIO can't be built there, and pip stops at the
# first failure, so nothing after this line gets installed either.
# (Found 2026-08-12 while preparing to install on Windows laptops.)
RPi.GPIO>=0.7.1; platform_machine in "armv6l armv7l aarch64"

# Lets the Launcher page find and stop programs quickly on any OS.
# Not strictly required — without it Windows falls back to calling
# PowerShell, which works but is much slower.
psutil>=5.9.0

# XML parser used by ohbot_pi.py to read motor calibration file.
# The 'ohbot' package only installs lxml on Windows and Mac, not Linux,
# so on the Pi it has to come from here.
lxml>=4.9.0
